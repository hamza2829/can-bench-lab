cmake_minimum_required(VERSION 3.20)
project(can_bench_lab VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CB_COVERAGE "Instrument the C core with gcov" OFF)
option(CB_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
include(FetchContent)
FetchContent_Declare(unity GIT_REPOSITORY https://github.com/ThrowTheSwitch/Unity.git GIT_TAG v2.6.1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(unity)
add_library(can_core core/frame.c core/memory_bus.c core/objects.c core/node.c)
target_include_directories(can_core PUBLIC include)
add_library(can_socketcan backends/socketcan.c)
target_link_libraries(can_socketcan PUBLIC can_core)
target_compile_definitions(can_socketcan PRIVATE _DEFAULT_SOURCE _POSIX_C_SOURCE=200809L)
add_executable(can_node app/main.c)
target_compile_definitions(can_node PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(can_node PRIVATE can_socketcan)
add_executable(test_core tests/c/test_core.c)
target_link_libraries(test_core PRIVATE can_core unity)
foreach(target can_core can_socketcan can_node test_core)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CB_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
if(CB_COVERAGE)
  target_compile_options(can_core PRIVATE --coverage -O0 -g)
  target_link_options(can_core INTERFACE --coverage)
endif()
enable_testing()
add_test(NAME core_unity COMMAND test_core)
